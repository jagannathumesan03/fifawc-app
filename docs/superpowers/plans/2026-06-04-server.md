# Server — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a local Node.js/Express server that handles room management, bracket submission, member bracket retrieval, and signed bracket photo export.

**Architecture:** Single Express app with in-memory storage (no database needed for local use). Rooms and brackets stored in Maps. Export endpoint generates a bracket image using `canvas` and signs it with HMAC-SHA256 using a server secret. Server runs on port 3000, matching `EXPO_PUBLIC_SERVER_URL` in `.env`.

**Tech Stack:** Node.js, Express, TypeScript, `canvas` (for image generation), `crypto` (built-in, for HMAC signing)

- [ ] **Step 4: Create `server/src/store.ts`**

```typescript
import crypto from 'crypto'
import type { Room, BracketSnapshot, StoredBracket } from './types'

function generateCode(): string {
  return Math.random().toString(36).slice(2, 6).toUpperCase()
}

function generateId(): string {
  return crypto.randomUUID()
}

function generateToken(): string {
  return crypto.randomBytes(24).toString('hex')
}

type StoreState = {
  rooms: Map<string, Room>
  brackets: Map<string, StoredBracket[]>
}

function createStore() {
  const state: StoreState = {
    rooms: new Map(),
    brackets: new Map(),
  }

  function usedCodes(): Set<string> {
    return new Set([...state.rooms.values()].map(r => r.code))
  }

  return {
    reset() {
      state.rooms.clear()
      state.brackets.clear()
    },

    createRoom(name: string, deviceId: string, displayName: string) {
      let code: string
      const used = usedCodes()
      do { code = generateCode() } while (used.has(code))

      const room: Room = {
        id: generateId(),
        code,
        name,
        adminId: deviceId,
        adminToken: generateToken(),
        status: 'lobby',
        members: [{ id: deviceId, displayName, hasSubmitted: false, submittedAt: null }],
        createdAt: new Date().toISOString(),
      }
      state.rooms.set(room.id, room)
      state.brackets.set(room.id, [])
      return { room, adminToken: room.adminToken }
    },

    joinRoom(code: string, deviceId: string, displayName: string): Room | null {
      const room = [...state.rooms.values()].find(r => r.code === code)
      if (!room) return null
      if (!room.members.find(m => m.id === deviceId)) {
        room.members.push({ id: deviceId, displayName, hasSubmitted: false, submittedAt: null })
      }
      return room
    },

    getRoom(roomId: string): Room | undefined {
      return state.rooms.get(roomId)
    },

    startRoom(roomId: string, adminToken: string): Room {
      const room = state.rooms.get(roomId)
      if (!room) throw new Error('Room not found')
      if (room.adminToken !== adminToken) throw new Error('Forbidden')
      room.status = 'active'
      return room
    },

    submitBracket(roomId: string, memberId: string, displayName: string, snapshot: BracketSnapshot) {
      const brackets = state.brackets.get(roomId) ?? []
      const existing = brackets.findIndex(b => b.memberId === memberId)
      const entry: StoredBracket = { memberId, displayName, snapshot, submittedAt: new Date().toISOString() }
      if (existing >= 0) brackets[existing] = entry
      else brackets.push(entry)
      state.brackets.set(roomId, brackets)

      const room = state.rooms.get(roomId)
      if (room) {
        const member = room.members.find(m => m.id === memberId)
        if (member) { member.hasSubmitted = true; member.submittedAt = entry.submittedAt }
      }
    },

    getMemberBrackets(roomId: string): StoredBracket[] {
      return state.brackets.get(roomId) ?? []
    },
  }
}

export const store = createStore()
```

```typescript
import { Router, Request, Response } from 'express'
import { store } from '../store'

const router = Router()

// POST /api/rooms — create room
router.post('/', (req: Request, res: Response) => {
  const { name, deviceId, displayName } = req.body
  if (!name || !deviceId || !displayName) return res.status(400).json({ error: 'name, deviceId, displayName required' })
  const { room, adminToken } = store.createRoom(name, deviceId, displayName)
  const { adminToken: _, ...publicRoom } = room
  res.status(201).json({ room: publicRoom, adminToken })
})

// POST /api/rooms/join
router.post('/join', (req: Request, res: Response) => {
  const { code, deviceId, displayName } = req.body
  if (!code || !deviceId || !displayName) return res.status(400).json({ error: 'code, deviceId, displayName required' })
  const room = store.joinRoom(code.toUpperCase(), deviceId, displayName)
  if (!room) return res.status(404).json({ error: 'Room not found' })
  const { adminToken: _, ...publicRoom } = room
  res.json({ room: publicRoom })
})

// GET /api/rooms/:id
router.get('/:id', (req: Request, res: Response) => {
  const room = store.getRoom(req.params.id)
  if (!room) return res.status(404).json({ error: 'Not found' })
  const { adminToken: _, ...publicRoom } = room
  res.json(publicRoom)
})

// POST /api/rooms/:id/start
router.post('/:id/start', (req: Request, res: Response) => {
  const token = req.headers.authorization?.replace('Bearer ', '') ?? ''
  try {
    const room = store.startRoom(req.params.id, token)
    const { adminToken: _, ...publicRoom } = room
    res.json({ room: publicRoom })
  } catch (e: unknown) {
    const msg = e instanceof Error ? e.message : 'Error'
    res.status(msg === 'Forbidden' ? 403 : 404).json({ error: msg })
  }
})

export default router
```

```typescript
import crypto from 'crypto'
import path from 'path'
import fs from 'fs'

const SERVER_SECRET = process.env.SERVER_SECRET ?? 'dev-secret-change-in-production'
const EXPORT_DIR = path.join(__dirname, '../../exports')

export function signBracket(deviceId: string, completedAt: string): string {
  return crypto
    .createHmac('sha256', SERVER_SECRET)
    .update(`${deviceId}:${completedAt}`)
    .digest('hex')
}

export async function generateBracketImage(
  deviceId: string,
  displayName: string,
  completedAt: string,
  signedKey: string
): Promise<string> {
  // Ensure exports directory exists
  if (!fs.existsSync(EXPORT_DIR)) fs.mkdirSync(EXPORT_DIR, { recursive: true })

  const { createCanvas } = await import('canvas')
  const canvas = createCanvas(800, 500)
  const ctx = canvas.getContext('2d')

  // Background
  ctx.fillStyle = '#0f172a'
  ctx.fillRect(0, 0, 800, 500)

  // Header bar
  ctx.fillStyle = '#22c55e'
  ctx.fillRect(0, 0, 800, 80)

  // Title
  ctx.fillStyle = '#ffffff'
  ctx.font = 'bold 28px sans-serif'
  ctx.fillText('FIFA World Cup 2026', 40, 50)

  // Subtitle
  ctx.fillStyle = '#86efac'
  ctx.font = '16px sans-serif'
  ctx.fillText('Bracket Prediction', 40, 70)

  // User info
  ctx.fillStyle = '#f1f5f9'
  ctx.font = 'bold 22px sans-serif'
  ctx.fillText(displayName, 40, 140)

  // Timestamp
  ctx.fillStyle = '#94a3b8'
  ctx.font = '14px sans-serif'
  ctx.fillText(`Predicted on: ${new Date(completedAt).toLocaleString()}`, 40, 170)

  // Signed key (truncated)
  ctx.fillStyle = '#475569'
  ctx.font = '11px monospace'
  ctx.fillText(`Key: ${signedKey.slice(0, 32)}…`, 40, 200)

  // Watermark
  ctx.fillStyle = '#1e293b'
  ctx.font = 'bold 14px sans-serif'
  ctx.fillText('fifawc-app · Tamper-proof prediction', 40, 480)

  const filename = `bracket-${deviceId}-${Date.now()}.png`
  const filepath = path.join(EXPORT_DIR, filename)
  const buffer = canvas.toBuffer('image/png')
  fs.writeFileSync(filepath, buffer)

  return `/exports/${filename}`
}
```

```typescript
import { Router, Request, Response } from 'express'
import { store } from '../store'
import { signBracket, generateBracketImage } from '../export'

const router = Router({ mergeParams: true })

// POST /api/rooms/:id/submit
router.post('/submit', (req: Request, res: Response) => {
  const { snapshot, deviceId } = req.body
  const room = store.getRoom(req.params.id)
  if (!room) return res.status(404).json({ error: 'Room not found' })
  const member = room.members.find(m => m.id === deviceId)
  if (!member) return res.status(403).json({ error: 'Not a member of this room' })
  store.submitBracket(req.params.id, deviceId, member.displayName, snapshot)
  res.status(204).send()
})

// GET /api/rooms/:id/brackets
router.get('/brackets', (req: Request, res: Response) => {
  const room = store.getRoom(req.params.id)
  if (!room) return res.status(404).json({ error: 'Room not found' })
  const brackets = store.getMemberBrackets(req.params.id)
  res.json(brackets)
})

// POST /api/rooms/:id/export
router.post('/export', async (req: Request, res: Response) => {
  const { snapshot, deviceId } = req.body
  const room = store.getRoom(req.params.id)
  if (!room) return res.status(404).json({ error: 'Room not found' })
  const member = room.members.find(m => m.id === deviceId)
  if (!member) return res.status(403).json({ error: 'Not a member' })

  const completedAt = snapshot.completedAt ?? new Date().toISOString()
  const signedKey = signBracket(deviceId, completedAt)

  try {
    const imagePath = await generateBracketImage(deviceId, member.displayName, completedAt, signedKey)
    const baseUrl = process.env.SERVER_URL ?? `http://localhost:${process.env.PORT ?? 3000}`
    res.json({ imageUrl: `${baseUrl}${imagePath}`, signedKey })
  } catch (e) {
    res.status(500).json({ error: 'Export failed' })
  }
})

export default router
```
